cmake_minimum_required(VERSION 3.22.1)
project(index_probe LANGUAGES C CXX)
# Keep CPU inference optimized even when the Flutter UI uses a debug build.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_CXX_STANDARD 17)
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
  message(FATAL_ERROR "A verified whisper.cpp checkout is required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
# Android x86_64 emulator is a CI target, without assuming host AVX support.
if(ANDROID_ABI STREQUAL "x86_64")
  foreach(feature AVX AVX2 AVX512 FMA F16C)
    set(GGML_${feature} OFF CACHE BOOL "" FORCE)
  endforeach()
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
add_library(index_probe SHARED probe.cpp)
target_link_libraries(index_probe PRIVATE whisper)
